cmake_minimum_required(VERSION 3.16)
project(psbbv_system_preferences C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(psbbv_preferences src/preferences.c)
target_include_directories(psbbv_preferences PUBLIC include)
target_compile_options(psbbv_preferences PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_executable(test_preferences tests/test_preferences.c)
target_link_libraries(test_preferences PRIVATE psbbv_preferences)
target_compile_options(test_preferences PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME preferences COMMAND test_preferences)
